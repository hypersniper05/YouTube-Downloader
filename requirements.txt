# YouTube to MP3/MP4 Converter with Whisper Transcription
# For manual installation (Docker handles this automatically)

# YouTube extraction needs a JavaScript runtime plus the player-challenge solver.
# Without both, every download fails with "HTTP Error 403: Forbidden".
#   [default] pulls in yt-dlp-ejs (the solver, exact-pinned by yt-dlp)
#   [deno]    pulls in the Deno runtime as a platform-correct wheel
# Deno is preferred over Node because it sandboxes the untrusted player JS.
# If you already have Deno or Node on PATH, plain yt-dlp[default] is enough.
yt-dlp[default,deno]>=2026.8.19

torch
transformers>=4.40.0
accelerate

# UVR vocal separation (MDX-Net models). With an NVIDIA GPU, install onnxruntime-gpu instead.
onnxruntime

# Note: You also need ffmpeg installed on your system
# Ubuntu/Debian: sudo apt install ffmpeg
# macOS: brew install ffmpeg
# Windows: winget install FFmpeg
